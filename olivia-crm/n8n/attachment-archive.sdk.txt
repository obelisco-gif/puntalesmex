import { workflow, node, trigger, expr, newCredential } from '@n8n/workflow-sdk';
const schedule=trigger({type:'n8n-nodes-base.scheduleTrigger',version:1.3,config:{name:'Revisar adjuntos cada minuto',parameters:{rule:{interval:[{field:'minutes',minutesInterval:1}]}}}});
const claim=node({type:'n8n-nodes-base.postgres',version:2.6,config:{name:'Reclamar adjunto',parameters:{operation:'executeQuery',query:'select * from public.olivia_claim_attachment();'},credentials:{postgres:newCredential('Postgres Sadiet')}}});
const get=node({type:'n8n-nodes-base.whatsApp',version:1.1,config:{name:'Obtener URL temporal',parameters:{resource:'media',operation:'mediaUrlGet',mediaGetId:expr('{{ $json.media_id }}')},credentials:{whatsAppApi:newCredential('Cafe WhatsApp account')}}});
const validate=node({type:'n8n-nodes-base.code',version:2,config:{name:'Validar tamaño y tipo',parameters:{mode:'runOnceForEachItem',jsCode:"const expected=$('Reclamar adjunto').item.json; if(!Number.isFinite(Number($json.file_size)) || Number($json.file_size)>20971520 || String($json.mime_type||'').split(';')[0]!==expected.mime_type) return null; return {json:$json};"}}});
const download=node({type:'n8n-nodes-base.httpRequest',version:4.4,config:{name:'Descargar adjunto',parameters:{url:expr('{{ $json.url }}'),authentication:'predefinedCredentialType',nodeCredentialType:'whatsAppApi',options:{response:{response:{responseFormat:'file'}},timeout:30000}},credentials:{whatsAppApi:newCredential('Cafe WhatsApp account')}}});
const encode=node({type:'n8n-nodes-base.code',version:2,config:{name:'Preparar archivo privado',parameters:{mode:'runOnceForAllItems',jsCode:"const b=await this.helpers.getBinaryDataBuffer(0,'data'); if(b.length>20971520) throw new Error('Adjunto demasiado grande'); const p=$('Reclamar adjunto').first().json; return [{json:{...p,data_b64:b.toString('base64')}}];"}}});
const save=node({type:'n8n-nodes-base.postgres',version:2.6,config:{name:'Guardar adjunto privado',parameters:{operation:'executeQuery',query:'select public.olivia_save_message_attachment($1,$2,$3,$4) as result;',options:{queryReplacement:expr('{{ [ $json.external_message_id, $json.media_id, $json.mime_type, $json.data_b64 ] }}')}},credentials:{postgres:newCredential('Postgres Sadiet')}}});
export default workflow('olivia-archivar-adjuntos','Olivia · Archivar adjuntos').add(schedule).to(claim).to(get).to(validate).to(download).to(encode).to(save);
